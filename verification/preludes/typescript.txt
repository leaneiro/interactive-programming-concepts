declare const console: { log(output: string): void };

class TreeNode {
  value: number;
  left: TreeNode | null = null;
  right: TreeNode | null = null;

  constructor(value: number) {
    this.value = value;
  }
}

interface Graph {
  edges: Map<string, string[]>;
}

function buildTree(values: number[]): TreeNode | null {
  let root: TreeNode | null = null;
  for (const value of values) {
    root = insertNode(root, value);
  }
  return root;
}

function insertNode(node: TreeNode | null, value: number): TreeNode {
  if (node === null) {
    return new TreeNode(value);
  }
  if (value < node.value) {
    node.left = insertNode(node.left, value);
  } else if (value > node.value) {
    node.right = insertNode(node.right, value);
  }
  return node;
}

function listInOrder(node: TreeNode | null): number[] {
  if (node === null) {
    return [];
  }
  return [...listInOrder(node.left), node.value, ...listInOrder(node.right)];
}
