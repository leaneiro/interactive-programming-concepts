#include <iostream>
#include <utility>
#include <vector>

using namespace std;

struct TreeNode {
    int value;
    TreeNode* left = nullptr;
    TreeNode* right = nullptr;

    explicit TreeNode(int value) : value(value) {}
};

TreeNode* insertNode(TreeNode* node, int value) {
    if (node == nullptr) {
        return new TreeNode(value);
    }
    if (value < node->value) {
        node->left = insertNode(node->left, value);
    } else if (value > node->value) {
        node->right = insertNode(node->right, value);
    }
    return node;
}

TreeNode* buildTree(const vector<int>& values) {
    TreeNode* root = nullptr;
    for (int value : values) {
        root = insertNode(root, value);
    }
    return root;
}

void collectInOrder(TreeNode* node, vector<int>& values) {
    if (node == nullptr) {
        return;
    }
    collectInOrder(node->left, values);
    values.push_back(node->value);
    collectInOrder(node->right, values);
}

vector<int> listInOrder(TreeNode* node) {
    vector<int> values;
    collectInOrder(node, values);
    return values;
}

void printResult(bool result) {
    cout << (result ? "true" : "false") << endl;
}

void printResult(const vector<int>& values) {
    cout << "[";
    for (size_t position = 0; position < values.size(); position++) {
        if (position > 0) {
            cout << ",";
        }
        cout << values[position];
    }
    cout << "]" << endl;
}
