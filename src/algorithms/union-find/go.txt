func unionFind(n int, pairs [][2]int) []int {
	parent := make([]int, n)
	for i := range parent {
		parent[i] = i
	}
	for _, pair := range pairs {
		union(parent, pair[0], pair[1])
	}
	return parent
}

func find(parent []int, x int) int {
	if parent[x] != x {
		parent[x] = find(parent, parent[x])
	}
	return parent[x]
}

func union(parent []int, a, b int) {
	rootA := find(parent, a)
	rootB := find(parent, b)
	if rootA != rootB {
		parent[rootB] = rootA
	}
}
