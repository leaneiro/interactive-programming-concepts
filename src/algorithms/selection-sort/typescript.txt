function selectionSort(arr: number[]): number[] {
  const a = arr.slice();
  for (let i = 0; i < a.length - 1; i++) {
    let smallest = i;
    for (let j = i + 1; j < a.length; j++) {
      if (a[j] < a[smallest]) {
        smallest = j;
      }
    }
    if (smallest !== i) {
      const tmp = a[i];
      a[i] = a[smallest];
      a[smallest] = tmp;
    }
  }
  return a;
}
