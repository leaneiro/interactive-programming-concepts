fun selectionSort(a: IntArray): IntArray {
    for (i in 0 until a.size - 1) {
        var smallest = i
        for (j in i + 1 until a.size) {
            if (a[j] < a[smallest]) {
                smallest = j
            }
        }
        if (smallest != i) {
            val tmp = a[i]
            a[i] = a[smallest]
            a[smallest] = tmp
        }
    }
    return a
}
