vector<int> selectionSort(vector<int> a) {
    int n = a.size();
    for (int i = 0; i < n - 1; i++) {
        int smallest = i;
        for (int j = i + 1; j < n; j++) {
            if (a[j] < a[smallest]) {
                smallest = j;
            }
        }
        if (smallest != i) {
            swap(a[i], a[smallest]);
        }
    }
    return a;
}
