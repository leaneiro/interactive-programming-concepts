int[] selectionSort(int[] a) {
    for (int i = 0; i < a.length - 1; i++) {
        int smallest = i;
        for (int j = i + 1; j < a.length; j++) {
            if (a[j] < a[smallest]) {
                smallest = j;
            }
        }
        if (smallest != i) {
            int tmp = a[i];
            a[i] = a[smallest];
            a[smallest] = tmp;
        }
    }
    return a;
}
