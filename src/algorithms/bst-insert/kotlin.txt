fun bstInsert(root: TreeNode?, value: Int): TreeNode {
    if (root == null) {
        return TreeNode(value)
    }

    if (value < root.value) {
        root.left = bstInsert(root.left, value)
    } else if (value > root.value) {
        root.right = bstInsert(root.right, value)
    }

    return root
}
