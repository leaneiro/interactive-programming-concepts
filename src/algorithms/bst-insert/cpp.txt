TreeNode* bstInsert(TreeNode* root, int value) {
    if (root == nullptr) {
        return new TreeNode(value);
    }

    if (value < root->value) {
        root->left = bstInsert(root->left, value);
    } else if (value > root->value) {
        root->right = bstInsert(root->right, value);
    }

    return root;
}
