func bstDelete(root *TreeNode, value int) *TreeNode {
	if root == nil {
		return nil
	}

	if value < root.value {
		root.left = bstDelete(root.left, value)
	} else if value > root.value {
		root.right = bstDelete(root.right, value)
	} else { // Node found: handle 3 cases
		// Case 1: Leaf node
		if root.left == nil && root.right == nil {
			return nil
		}
		// Case 2: One child
		if root.left == nil {
			return root.right
		}
		if root.right == nil {
			return root.left
		}
		// Case 3: Two children - Find min in right subtree
		successor := root.right
		for successor.left != nil {
			successor = successor.left
		}

		root.value = successor.value
		root.right = bstDelete(root.right, successor.value)
	}
	return root
}
