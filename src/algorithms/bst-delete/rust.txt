fn bst_delete(root: Option<Box<TreeNode>>, value: i32) -> Option<Box<TreeNode>> {
    let mut root = root?;

    if value < root.value {
        root.left = bst_delete(root.left.take(), value);
    } else if value > root.value {
        root.right = bst_delete(root.right.take(), value);
    } else {
        // Node found: handle 3 cases
        // Case 1: Leaf node
        if root.left.is_none() && root.right.is_none() {
            return None;
        }
        // Case 2: One child
        if root.left.is_none() {
            return root.right;
        }
        if root.right.is_none() {
            return root.left;
        }
        // Case 3: Two children - Find min in right subtree
        let mut successor = root.right.as_ref().unwrap();
        while let Some(left) = successor.left.as_ref() {
            successor = left;
        }

        root.value = successor.value;
        root.right = bst_delete(root.right.take(), root.value);
    }
    Some(root)
}
