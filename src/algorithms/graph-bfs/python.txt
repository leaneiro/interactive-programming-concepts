def bfs(graph, start):
    visited = set()
    queue = deque()
    result = []

    visited.add(start)
    queue.append(start)

    while queue:
        current = queue.popleft()
        result.append(current)

        for neighbor in graph[current]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)
    return result
